class Solution {
  public:
    int minToggle(vector<int>& arr) {
        int n = arr.size();
        
        int ones = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 1) ones++;
        }
        int zeros = n - ones;
      
        int ans = zeros;
        
        int leftOnes = 0; 
        int rightZeros = zeros; 
        
        for (int i = 0; i < n; i++) {
            if (arr[i] == 1) {
                leftOnes++;
            } else {
                rightZeros--;
            }
            int cost = leftOnes + rightZeros;
            ans = min(ans, cost);
        }
        
        return ans;
    }
};
