class Solution {
  public:
    int findIndex(string &s) {
        int n = s.size();
        
        vector<int> open(n + 1, 0);
        vector<int> close(n + 1, 0);
        
        for (int i = 0; i < n; i++)
            open[i + 1] = open[i] + (s[i] == '(' ? 1 : 0);
        
        for (int i = n - 1; i >= 0; i--)
            close[i] = close[i + 1] + (s[i] == ')' ? 1 : 0);
        
        for (int k = 0; k <= n; k++)
            if (open[k] == close[k])
                return k;
        
        return -1;
    }
};
